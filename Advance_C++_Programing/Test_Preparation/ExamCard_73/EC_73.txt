73. Отношения между концептами.
Уточнение (refinement):

cpp
template <typename T>
concept SignedIntegral = std::integral<T> && std::is_signed_v<T>;
Объединение:

cpp
template <typename T>
concept Arithmetic = std::integral<T> || std::floating_point<T>;
Отрицание:

cpp
template <typename T>
concept NotPointer = !std::is_pointer_v<T>;
