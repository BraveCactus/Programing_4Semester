class Shape {
public:
    virtual void draw() = 0;
};

class Circle : public Shape { /*...*/ };
class Square : public Shape { /*...*/ };

class ShapeFactory {
public:
    static Shape* create(const std::string& type) {
        if (type == "circle") return new Circle();
        if (type == "square") return new Square();
        return nullptr;
    }
};
